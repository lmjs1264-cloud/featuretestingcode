===== FILE 1: lib/referrals.js =====
// referrals.js — put in lib/ in Base44.
// Decides what referral content may be shown where, and shares invite links.
//
//   "web"       website: show all referral offers as today (Stripe).
//   "app-us"    iPhone/Android app on the US store: referral pricing can't be
//               bought in the app, but Apple allows pointing US users to the
//               website, so show a "claim it on glymdot.com" note + link.
//   "app-other" app on any other country's store: Apple forbids mentioning
//               outside offers, so hide referral pricing completely.
// In every mode people can still SEND their referral link.

import { isAppStoreBuild, setupPurchases } from "./inAppPurchases";

const plugins = () => window?.Capacitor?.Plugins || {};

let cachedMode = null;

/** Resolves to "web" | "app-us" | "app-other". Pass the logged-in user's id. */
export async function getReferralMode(userId) {
  if (!isAppStoreBuild()) return "web";
  if (cachedMode) return cachedMode;
  try {
    await setupPurchases(userId);
    const { countryCode } = await plugins().Purchases.getStorefront();
    // iPhone reports "USA", Android reports "US".
    cachedMode = ["USA", "US"].includes(String(countryCode).toUpperCase()) ? "app-us" : "app-other";
  } catch {
    // Can't tell the country: take the safe option and say nothing about offers.
    cachedMode = "app-other";
  }
  return cachedMode;
}

/** May referral pricing / extended trials be described here? */
export const canShowReferralOffers = (mode) => mode === "web" || mode === "app-us";

/**
 * Opens the phone's share sheet (Messages, WhatsApp, email…) with the invite.
 * Falls back to copying the link. Returns "shared" | "copied" | "cancelled".
 */
export async function shareReferralLink({ url, text = "Join me on glym", title = "glym" }) {
  const Share = plugins().Share;
  try {
    if (isAppStoreBuild() && Share) {
      await Share.share({ title, text, url, dialogTitle: "Invite to glym" });
      return "shared";
    }
    if (navigator.share) {
      await navigator.share({ title, text, url });
      return "shared";
    }
  } catch (e) {
    if (/cancel|abort/i.test(String(e?.message || e?.name || e))) return "cancelled";
    // anything else: fall through to copying
  }
  await copyText(url);
  return "copied";
}

async function copyText(value) {
  try {
    await navigator.clipboard.writeText(value);
  } catch {
    const box = document.createElement("textarea");
    box.value = value;
    box.setAttribute("readonly", "");
    box.style.position = "fixed";
    box.style.opacity = "0";
    document.body.appendChild(box);
    box.select();
    document.execCommand("copy");
    box.remove();
  }
}

/** Opens a glymdot.com page in Safari/Chrome (outside the app). */
export async function openInBrowser(url) {
  const Launcher = plugins().AppLauncher;
  if (isAppStoreBuild() && Launcher) {
    await Launcher.openUrl({ url });
  } else {
    window.open(url, "_blank", "noopener");
  }
}

// ReferralComponents.jsx — put in components/ in Base44.
//
// <ShareReferralButton url={user's referral link} />
//    The "invite a friend" button. Works on the website and in the app.
//
// <ReferralWebsiteNotice userId={user.id} claimUrl="https://glymdot.com/..." />
//    Put on the in-app upgrade screen. Shows the "claim on glymdot.com" note
//    only for US App Store users; shows nothing anywhere else.
//
// useReferralMode(userId)  -> "web" | "app-us" | "app-other" | null (still checking)
//    Wrap any referral pricing text with canShowReferralOffers(mode).

import React, { useEffect, useState } from "react";
import { getReferralMode, shareReferralLink, openInBrowser } from "../lib/referrals";

export function useReferralMode(userId) {
  const [mode, setMode] = useState(null);
  useEffect(() => {
    let alive = true;
    getReferralMode(userId).then((m) => alive && setMode(m));
    return () => { alive = false; };
  }, [userId]);
  return mode;
}

export function ShareReferralButton({ url, text, className = "", label = "Invite friends" }) {
  const [status, setStatus] = useState("");
  const share = async () => {
    setStatus("");
    const result = await shareReferralLink({ url, text });
    if (result === "copied") setStatus("Link copied. Paste it anywhere to share.");
  };
  return (
    <div className={className}>
      <button
        type="button"
        onClick={share}
        disabled={!url}
        className="w-full rounded-lg bg-blue-600 text-white px-4 py-2.5 font-medium disabled:opacity-50"
      >
        {label}
      </button>
      {status && <p className="mt-2 text-sm text-gray-600 dark:text-gray-300">{status}</p>}
    </div>
  );
}

export function ReferralWebsiteNotice({ userId, claimUrl, className = "" }) {
  const mode = useReferralMode(userId);
  if (mode !== "app-us") return null;
  return (
    <div className={`rounded-xl border border-gray-200 dark:border-gray-700 p-4 ${className}`}>
      <p className="text-sm">
        Were you invited by a friend? Referral offers are available on glymdot.com.
        Sign in there with this same account and your Premium works here too.
      </p>
      <button
        type="button"
        onClick={() => openInBrowser(claimUrl)}
        className="mt-3 w-full rounded-lg border border-gray-300 dark:border-gray-600 px-4 py-2 text-sm font-medium"
      >
        Claim referral offer on glymdot.com
      </button>
    </div>
  );
}
